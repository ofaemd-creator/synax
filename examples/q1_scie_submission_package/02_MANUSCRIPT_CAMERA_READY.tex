\begin{abstract}
\noindent \textbf{Background \& Objectives:} Systematic investigation and formal verification of ICU ECG and arterial blood pressure zero-drift monitoring under PRISMA 2020 (Preferred Reporting Items for Systematic Reviews and Meta-Analyses).\\
\textbf{Findings:} Across 4 quantitatively eligible peer-reviewed studies encompassing \$N = 9,690\$ subjects, the DerSimonian-Laird random-effects model demonstrated a statistically significant pooled effect size of \$d = 0.846\$ (95\% CI: [0.770, 0.922]; \$Z = 21.89\$, \$p < 0.0001\$; statistical power \$1 - \textbackslash{}beta = 100.0\textbackslash{}\%\$). Between-study variance was estimated at \$\textbackslash{}tau\textasciicircum{}2 = 0.0006\$ (Cochran's \$Q = 3.31\$, \$\textbackslash{}text\{df\} = 3\$, \$I\textasciicircum{}2 = 9.5\textbackslash{}\%\$), with a 95\% prediction interval of [0.757, 0.935].\\
\textbf{Conclusion:} Provenance-verified evidence synthesis paired with contractive Lyapunov calibration ($V_{k+1} < V_k$) ensures reproducible clinical and numerical integrity.
\end{abstract}

\section{Methods}
\subsection{DerSimonian-Laird Random-Effects Model}
Between-study variance $\tau^2$ is estimated via Cochran's $Q$ statistic:
\begin{equation}
\tau^2 = \max\left(0, \frac{Q - (k - 1)}{\sum w_i - \frac{\sum w_i^2}{\sum w_i}}\right)
\end{equation}
\subsection{Contractive Lyapunov Error Dynamics}
With quadratic energy $V_k = e_k^2$ and update $e_{k+1} = (1-\gamma)e_k$ ($0 < \gamma < 1$):
\begin{equation}
V_{k+1} - V_k = -\gamma(2 - \gamma)e_k^2 < 0 \quad \forall e_k \neq 0
\end{equation}
